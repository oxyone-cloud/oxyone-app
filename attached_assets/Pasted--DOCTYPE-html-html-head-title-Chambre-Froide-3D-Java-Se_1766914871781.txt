    <script>
        const scene = new THREE.Scene();
        scene.background = new THREE.Color(0xf4f7f9);

        const camera = new THREE.PerspectiveCamera(45, window.innerWidth / window.innerHeight, 0.1, 1000);
        camera.position.set(7, 5, 8);

        const renderer = new THREE.WebGLRenderer({ antialias: true });
        renderer.setSize(window.innerWidth, window.innerHeight);
        document.body.appendChild(renderer.domElement);

        // Lumières "Studio" pour un rendu clair
        scene.add(new THREE.AmbientLight(0xffffff, 0.8));
        const spot = new THREE.DirectionalLight(0xffffff, 0.6);
        spot.position.set(5, 10, 5);
        scene.add(spot);

        // Construction de la chambre (Panneaux blancs)
        const panelMat = new THREE.MeshPhongMaterial({ color: 0xffffff });
        const blueFrameMat = new THREE.MeshPhongMaterial({ color: 0xadd8e6 });

        function addPanel(w, h, d, x, y, z) {
            const mesh = new THREE.Mesh(new THREE.BoxGeometry(w, h, d), panelMat);
            mesh.position.set(x, y, z);
            scene.add(mesh);
            
            // Bordures bleues (Profilés)
            const edges = new THREE.EdgesGeometry(new THREE.BoxGeometry(w, h, d));
            const line = new THREE.LineSegments(edges, new THREE.LineBasicMaterial({ color: 0x3498db }));
            line.position.set(x, y, z);
            scene.add(line);
        }

        // Dimensions conformes à ton image
        addPanel(2.74, 0.08, 1.83, 0, 0, 0);     // Sol
        addPanel(0.08, 2, 1.83, -1.37, 1, 0);   // Mur Gauche
        addPanel(2.74, 2, 0.08, 0, 1, -0.91);   // Mur Arrière
        addPanel(2.74, 0.08, 1.83, 0, 2.5, 0);  // Toit (Amovible/Éclaté)

        // Moteur de refroidissement sur le toit
        const motor = new THREE.Mesh(new THREE.BoxGeometry(0.7, 0.6, 0.6), new THREE.MeshPhongMaterial({color: 0xecf0f1}));
        motor.position.set(0, 3, 0);
        scene.add(motor);

        const controls = new THREE.OrbitControls(camera, renderer.domElement);
        controls.enableDamping = true;

        function animate() {
            requestAnimationFrame(animate);
            controls.update();
            renderer.render(scene, camera);
        }
        animate();
    </script>
