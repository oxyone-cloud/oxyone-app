<script>
// Exact configuration (utilise tes valeurs)
const config = {
  internal: {L:2.74, W:1.83, H:2.00}, // meters
  panelThickness_m: 0.080, // 80 mm
  doorThickness_m: 0.060, // 60 mm
  doorHeight_m: 0.80,
  ambientTemp:25,
  storageTempDefault:4,
  storageRange:[1,10],
  loadDensity_kg_m2:250
}

// We'll map real-world meters to pixels for display while preserving proportions.
// Choose a scale so the largest internal dimension fits comfortably.
const PIXELS_PER_M = 110; // 110 px = 1 m (adjust if needed)

// Compute pixel sizes
const innerPx = {
  width: Math.round(config.internal.W * PIXELS_PER_M),
  height: Math.round(config.internal.H * PIXELS_PER_M),
  length: Math.round(config.internal.L * PIXELS_PER_M)
};
const panelPx = Math.round(config.panelThickness_m * PIXELS_PER_M);
const doorPx = Math.round(config.doorHeight_m * PIXELS_PER_M);

// Create shell and inner elements programmatically with precise dimensions
const shell = document.getElementById('shell');
const inner = document.getElementById('inner');

function makeFace(className, w, h, styles = {}){
  const el = document.createElement('div');
  el.className = 'face ' + className;
  el.style.width = w + 'px';
  el.style.height = h + 'px';
  Object.keys(styles).forEach(k=>el.style[k]=styles[k]);
  return el;
}

function buildScene(){
  // Clear
  shell.innerHTML=''; inner.innerHTML='';

  // Outer shell represents internal dims + 2*thickness (for visualizing panel thickness)
  const totalWidth = innerPx.width + 2*panelPx;
  const totalHeight = innerPx.height + 2*panelPx;
  const totalLength = innerPx.length + 2*panelPx;

  // Position shell centrally using absolute positioning and transforms
  shell.style.width = totalLength + 'px';
  shell.style.height = totalHeight + 'px';
  shell.style.left = '50%'; shell.style.top = '60px'; shell.style.transform = 'translateX(-50%)';

  // Front face (outer)
  const front = makeFace('front', totalLength, totalHeight, {transform:`translateZ(${panelPx}px)`});
  const back  = makeFace('back', totalLength, totalHeight, {transform:`rotateY(180deg) translateZ(${panelPx}px)`});
  const left  = makeFace('left', panelPx, totalHeight, {transform:`rotateY(-90deg) translateZ(${(totalLength/2 - panelPx)}px)`, left: (-(panelPx))+'px'});
  const right = makeFace('right', panelPx, totalHeight, {transform:`rotateY(90deg) translateZ(${(totalLength/2 - panelPx)}px)`, left: (totalLength)+'px'});
  const top   = makeFace('top', totalLength, panelPx, {transform:`rotateX(90deg) translateZ(${(totalHeight/2 - panelPx)}px)`, top: (-(panelPx))+'px'});
  const bottom= makeFace('bottom', totalLength, panelPx, {transform:`rotateX(-90deg) translateZ(${(totalHeight/2 - panelPx)}px)`, top: (totalHeight)+'px'});

  shell.appendChild(front);
  shell.appendChild(back);
  shell.appendChild(left);
  shell.appendChild(right);
  shell.appendChild(top);
  shell.appendChild(bottom);

  // INNER - exact internal dimensions
  inner.style.width = innerPx.length + 'px';
  inner.style.height = innerPx.height + 'px';
  inner.style.left = '50%'; inner.style.top = (60 + panelPx) + 'px';
  inner.style.transform = 'translateX(-50%)';

  const inFront = document.createElement('div'); inFront.className='panel in-front'; inFront.style.width = innerPx.length+'px'; inFront.style.height = innerPx.height+'px'; inFront.style.transform = `translateZ(${panelPx}px)`;
  const inBack  = document.createElement('div'); inBack.className='panel in-back'; inBack.style.width = innerPx.length+'px'; inBack.style.height = innerPx.height+'px'; inBack.style.transform = `rotateY(180deg) translateZ(${panelPx}px)`;
  const inLeft  = document.createElement('div'); inLeft.className='panel in-left'; inLeft.style.width = panelPx+'px'; inLeft.style.height = innerPx.height+'px'; inLeft.style.left = (-panelPx)+'px'; inLeft.style.transform = `rotateY(-90deg) translateZ(${(innerPx.length/2)}px)`;
  const inRight = document.createElement('div'); inRight.className='panel in-right'; inRight.style.width = panelPx+'px'; inRight.style.height = innerPx.height+'px'; inRight.style.left = innerPx.length+'px'; inRight.style.transform = `rotateY(90deg) translateZ(${(innerPx.length/2)}px)`;
  const inTop   = document.createElement('div'); inTop.className='panel in-top'; inTop.style.width = innerPx.length+'px'; inTop.style.height = panelPx+'px'; inTop.style.top = (-panelPx)+'px'; inTop.style.transform = `rotateX(90deg) translateZ(${(innerPx.height/2)}px)`;
  const inBottom= document.createElement('div'); inBottom.className='panel in-bottom'; inBottom.style.width = innerPx.length+'px'; inBottom.style.height = panelPx+'px'; inBottom.style.top = innerPx.height+'px'; inBottom.style.transform = `rotateX(-90deg) translateZ(${(innerPx.height/2)}px)`;

  inner.appendChild(inFront); inner.appendChild(inBack); inner.appendChild(inLeft); inner.appendChild(inRight); inner.appendChild(inTop); inner.appendChild(inBottom);

  // Door: place on front inner face, right side, with exact height
  const doorEl = document.createElement('div');
  doorEl.className = 'door';
  const doorWidthPx = Math.round((config.internal.L * PIXELS_PER_M) * 0.25); // proportional width ~25% of length
  doorEl.style.width = doorWidthPx+'px';
  doorEl.style.height = doorPx+'px';
  // position relative to inner box
  const innerLeft = (window.innerWidth/2) - (innerPx.length/2);
  // We'll position by CSS within inner using absolute coords
  doorEl.style.right = (panelPx + 26) + 'px';
  doorEl.style.bottom = (panelPx + 20) + 'px';
  doorEl.innerHTML = `<div class="heightmark">H: ${config.doorHeight_m.toFixed(2)} m</div>`;
  inner.appendChild(doorEl);
}

// Steps and controls (same logic as before but adapted)
const steps = [
  "Étape 1 : Positionnement des panneaux (réception des modules)",
  "Étape 2 : Montage des parois latérales et alignement",
  "Étape 3 : Fixation du plancher et du toit",
  "Étape 4 : Installation de la porte amovible (ép. 60 mm)",
  "Étape 5 : Installation du groupe froid et tests électriques",
  "Étape 6 : Test final, contrôle température & charge"
];
let step = 0; const stepText = document.getElementById('stepText');
function applyStep(s){ step = s; stepText.innerText = steps[step]; document.getElementById('scene').style.transform='rotateX(6deg) rotateY(-22deg) translateZ(0)';
  // Visual hints: exploded on step 1 only for demo
  if(step===1){ document.getElementById('explodeBtn').classList.add('active'); }
}

// Buttons
document.getElementById('stepNext').addEventListener('click', ()=>{ if(step<steps.length-1) applyStep(step+1); });
document.getElementById('stepPrev').addEventListener('click', ()=>{ if(step>0) applyStep(step-1); });

// Exploded
let exploded=false; document.getElementById('explodeBtn').addEventListener('click', ()=>{ exploded=!exploded; if(exploded){ document.getElementById('shell').classList.add('exploded'); document.getElementById('inner').classList.add('exploded'); } else { document.getElementById('shell').classList.remove('exploded'); document.getElementById('inner').classList.remove('exploded'); } });

// Door open
let doorOpen=false; document.getElementById('doorBtn').addEventListener('click', ()=>{ doorOpen=!doorOpen; const d = document.querySelector('#inner .door'); if(doorOpen) d.classList.add('open'); else d.classList.remove('open'); });

// Auto-play
let autoplay=null; document.getElementById('autoPlay').addEventListener('click', ()=>{ if(autoplay){ clearInterval(autoplay); autoplay=null; document.getElementById('autoPlay').innerText='Auto-play'; return; } document.getElementById('autoPlay').innerText='Stop'; let i=0; applyStep(0); autoplay=setInterval(()=>{ i++; if(i>steps.length-1){ clearInterval(autoplay); autoplay=null; document.getElementById('autoPlay').innerText='Auto-play'; return; } applyStep(i); },1400); });

document.getElementById('resetBtn').addEventListener('click', ()=>{ applyStep(0); document.getElementById('shell').classList.remove('exploded'); document.getElementById('inner').classList.remove('exploded'); const d=document.querySelector('#inner .door'); if(d) d.classList.remove('open'); });

// Init
buildScene(); applyStep(0);

// Accessibility: expose computed sizes in console for verification
console.log('Internal (m):',config.internal);
console.log('Pixels per m:', PIXELS_PER_M);
console.log('Inner px sizes:', innerPx);
console.log('Panel px thickness:', panelPx, 'Door height px:', doorPx);

</script>
