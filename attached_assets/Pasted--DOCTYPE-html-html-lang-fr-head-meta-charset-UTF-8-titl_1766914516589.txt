<script type="importmap">
  {
    "imports": {
      "three": "https://unpkg.com/three@0.160.0/build/three.module.js",
      "three/addons/": "https://unpkg.com/three@0.160.0/examples/jsm/"
    }
  }
</script>

<script type="module">
    import * as THREE from 'three';
    import { OrbitControls } from 'three/addons/controls/OrbitControls.js';

    // 1. SCÈNE & CAMÉRA
    const scene = new THREE.Scene();
    scene.background = new THREE.Color(0xe1e5ea);
    const camera = new THREE.PerspectiveCamera(45, window.innerWidth / window.innerHeight, 0.1, 1000);
    camera.position.set(5, 5, 5);

    const renderer = new THREE.WebGLRenderer({ antialias: true });
    renderer.setSize(window.innerWidth, window.innerHeight);
    document.body.appendChild(renderer.domElement);

    // 2. LUMIÈRES (Style Laboratoire)
    const ambientLight = new THREE.AmbientLight(0xffffff, 0.7);
    scene.add(ambientLight);
    const dirLight = new THREE.DirectionalLight(0xffffff, 0.8);
    dirLight.position.set(5, 10, 7);
    scene.add(dirLight);

    // 3. CONSTRUCTION DE LA CHAMBRE (Dimensions de l'image)
    const L = 2.74, H = 2.00, W = 1.83;
    const thickness = 0.08; // 80mm comme sur l'image

    // Groupe pour la chambre froide
    const coldRoom = new THREE.Group();

    // Matériaux
    const panelMat = new THREE.MeshPhongMaterial({ color: 0xffffff });
    const frameMat = new THREE.MeshPhongMaterial({ color: 0xa0c4ff }); // Bleu clair de l'image

    // Fonction pour créer un mur
    function createWall(w, h, d, x, y, z) {
        const geo = new THREE.BoxGeometry(w, h, d);
        const mesh = new THREE.Mesh(geo, panelMat);
        mesh.position.set(x, y, z);
        coldRoom.add(mesh);
        
        // Ajout de bordures (Edges) pour le réalisme
        const edges = new THREE.EdgesGeometry(geo);
        const line = new THREE.LineSegments(edges, new THREE.LineBasicMaterial({ color: 0x999999 }));
        line.position.set(x, y, z);
        coldRoom.add(line);
    }

    // Sol, Mur arrière, Mur gauche
    createWall(L, thickness, W, 0, 0, 0); // Sol
    createWall(thickness, H, W, -L/2, H/2, 0); // Mur Gauche
    createWall(L, H, thickness, 0, H/2, -W/2); // Mur Arrière
    
    // Le plafond (décalé pour l'effet "éclaté" comme sur la photo)
    const roof = new THREE.Mesh(new THREE.BoxGeometry(L, thickness, W), panelMat);
    roof.position.set(0, H + 0.5, 0); // Un peu surélevé
    coldRoom.add(roof);

    scene.add(coldRoom);

    // 4. AJOUT DU MOTEUR (Simplifié)
    const motorGeo = new THREE.BoxGeometry(0.6, 0.5, 0.4);
    const motor = new THREE.Mesh(motorGeo, new THREE.MeshPhongMaterial({color: 0xdddddd}));
    motor.position.set(0, H + 0.8, 0);
    scene.add(motor);

    // 5. CONTRÔLES
    const controls = new OrbitControls(camera, renderer.domElement);
    controls.enableDamping = true;

    // Animation
    function animate() {
        requestAnimationFrame(animate);
        controls.update();
        renderer.render(scene, camera);
    }
    animate();

    // Redimensionnement
    window.addEventListener('resize', () => {
        camera.aspect = window.innerWidth / window.innerHeight;
        camera.updateProjectionMatrix();
        renderer.setSize(window.innerWidth, window.innerHeight);
    });
</script>
